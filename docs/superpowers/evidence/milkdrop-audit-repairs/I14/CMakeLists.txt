cmake_minimum_required(VERSION 3.22)
project(motion_field_proposal LANGUAGES C CXX)
if(NOT PROJECTM_SOURCE OR NOT REPO_ROOT)
 message(FATAL_ERROR "Provide current frozen PROJECTM_SOURCE and REPO_ROOT")
endif()
add_subdirectory("${REPO_ROOT}/core/src/test/native/projectm-regressions" existing-controls)
add_executable(motion-field-controls motion_field_controls.cpp)
target_compile_features(motion-field-controls PRIVATE cxx_std_17)
target_include_directories(motion-field-controls PRIVATE "${REPO_ROOT}/core/src/test/native/projectm-regressions" "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor" "${PROJECTM_SOURCE}/vendor/glad/include" "${PROJECTM_SOURCE}/vendor/projectm-eval/projectm-eval/api")
target_link_libraries(motion-field-controls PRIVATE libprojectM::projectM)
if(APPLE)
 find_library(MOTION_OPENGL_FRAMEWORK OpenGL REQUIRED)
 target_link_libraries(motion-field-controls PRIVATE "${MOTION_OPENGL_FRAMEWORK}")
 target_compile_definitions(motion-field-controls PRIVATE GL_SILENCE_DEPRECATION)
 target_compile_options(motion-field-controls PRIVATE "-include${REPO_ROOT}/core/src/test/native/projectm-regressions/macos_gl.hpp")
else()
 message(FATAL_ERROR "This wrapper is CGL only; qualify target GLES separately")
endif()
if(SANITIZERS)
 target_compile_options(motion-field-controls PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(motion-field-controls PRIVATE -fsanitize=address,undefined)
endif()
