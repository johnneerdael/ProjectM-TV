cmake_minimum_required(VERSION 3.22)
project(motion_style_controls VERSION 4.2.0 LANGUAGES C CXX)
if(NOT APPLE)
    message(FATAL_ERROR "This proposal qualifies CGL only; GLES TF/tie-bias/cache behavior needs separate admission")
endif()
if(NOT PROJECTM_SOURCE)
    message(FATAL_ERROR "PROJECTM_SOURCE must point to the already-patched production engine")
endif()
set(REGRESSION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../core/src/test/native/projectm-regressions")
option(SANITIZERS "Use repository ASan/UBSan policy" ON)
add_subdirectory("${REGRESSION_ROOT}" existing-controls EXCLUDE_FROM_ALL)
add_executable(motion-style-controls motion_style_controls.cpp)
target_compile_features(motion-style-controls PRIVATE cxx_std_17)
target_compile_definitions(motion-style-controls PRIVATE GL_SILENCE_DEPRECATION)
target_include_directories(motion-style-controls PRIVATE "${REGRESSION_ROOT}"
    "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor"
    "${PROJECTM_SOURCE}/vendor/glad/include"
    "${PROJECTM_SOURCE}/vendor/projectm-eval/projectm-eval/api")
target_compile_options(motion-style-controls PRIVATE "-include${REGRESSION_ROOT}/macos_gl.hpp")
target_link_libraries(motion-style-controls PRIVATE libprojectM::projectM)
find_library(MOTION_STYLE_OPENGL_FRAMEWORK OpenGL REQUIRED)
target_link_libraries(motion-style-controls PRIVATE "${MOTION_STYLE_OPENGL_FRAMEWORK}")
if(SANITIZERS)
    target_compile_options(motion-style-controls PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(motion-style-controls PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME motion-style-controls COMMAND motion-style-controls)
set_tests_properties(motion-style-controls PROPERTIES TIMEOUT 90 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
