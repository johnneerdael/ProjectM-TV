cmake_minimum_required(VERSION 3.22)
project(i04_bits_integration_proposal LANGUAGES C CXX)
if(NOT PROJECTM_SOURCE OR NOT REPO_ROOT)
 message(FATAL_ERROR "Provide root-owned full patched PROJECTM_SOURCE and REPO_ROOT")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(PROJECTM_EVAL_FLOAT_SIZE 8 CACHE STRING "production double evaluator")
set(ENABLE_FAST_MATH ON CACHE BOOL "qualify release fast-math")
if(ANDROID AND NOT TARGET OpenGL::GLES3)
 add_library(OpenGL::GLES3 INTERFACE IMPORTED GLOBAL)
 set_property(TARGET OpenGL::GLES3 PROPERTY INTERFACE_LINK_LIBRARIES "EGL;GLESv3;log")
endif()
add_subdirectory("${REPO_ROOT}/core/src/test/native/projectm-regressions" actual-projectm-controls)
add_executable(eel-i04-bits-controls eel_i04_bits_test.cpp)
target_compile_features(eel-i04-bits-controls PRIVATE cxx_std_17)
target_include_directories(eel-i04-bits-controls PRIVATE "${PROJECTM_SOURCE}/vendor/projectm-eval/projectm-eval/api")
target_link_libraries(eel-i04-bits-controls PRIVATE libprojectM::projectM)
if(CMAKE_CXX_COMPILER_ID MATCHES "^(Clang|AppleClang|GNU)$")
 target_compile_options(eel-i04-bits-controls PRIVATE -fno-fast-math)
endif()
if(APPLE)
 find_library(ARITHMETIC_OPENGL_FRAMEWORK OpenGL REQUIRED)
 target_link_libraries(eel-i04-bits-controls PRIVATE "${ARITHMETIC_OPENGL_FRAMEWORK}")
endif()
if(SANITIZERS)
 target_compile_options(eel-i04-bits-controls PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(eel-i04-bits-controls PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(family I04 boundaries aliases)
 add_test(NAME eel-i04-bits-${family} COMMAND eel-i04-bits-controls ${family})
 set_tests_properties(eel-i04-bits-${family} PROPERTIES TIMEOUT 20 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endforeach()
