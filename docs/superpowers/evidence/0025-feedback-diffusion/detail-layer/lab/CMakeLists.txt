cmake_minimum_required(VERSION 3.22)
project(detail_combine_test LANGUAGES CXX)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(detail-combine-test combine-test.cpp)
target_compile_features(detail-combine-test PRIVATE cxx_std_17)
target_compile_definitions(detail-combine-test PRIVATE GL_SILENCE_DEPRECATION)
target_include_directories(detail-combine-test PRIVATE "${REPO_ROOT}/tools/preset-lab/src/preset_lab/native")
target_link_libraries(detail-combine-test PRIVATE SDL2::SDL2 OpenGL::GL)
