cmake_minimum_required(VERSION 3.22)
project(pmbench VERSION 4.1.7 LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL_UI OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_GLM OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_FLEX ON)
set(CMAKE_DISABLE_FIND_PACKAGE_BISON ON)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_CURRENT_SOURCE_DIR}/analysis_hooks.hpp>")
add_subdirectory("${PROJECTM_SOURCE}" projectm EXCLUDE_FROM_ALL)
add_executable(pmbench bench.cpp)
target_compile_features(pmbench PRIVATE cxx_std_17)
if(UPSTREAM)
  target_compile_definitions(pmbench PRIVATE PMBENCH_UPSTREAM)
endif()
target_include_directories(pmbench PRIVATE "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor/glm")
target_link_libraries(pmbench PRIVATE libprojectM::projectM EGL GLESv3 log)
target_link_options(pmbench PRIVATE -static-libstdc++)
