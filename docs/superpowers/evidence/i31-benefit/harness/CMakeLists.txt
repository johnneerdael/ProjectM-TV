cmake_minimum_required(VERSION 3.22)
project(preset_lab VERSION 4.2.0 LANGUAGES C CXX)
if(NOT PROJECTM_SOURCE)
    message(FATAL_ERROR "PROJECTM_SOURCE must identify the isolated instrumented engine")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL_UI OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_FLEX ON)
set(CMAKE_DISABLE_FIND_PACKAGE_BISON ON)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_CURRENT_SOURCE_DIR}/analysis_hooks.hpp>")
# Internal renderer headers use GLAD; load its declarations before forced GL shims.
include_directories("${PROJECTM_SOURCE}/vendor/glad/include")
add_subdirectory("${PROJECTM_SOURCE}" projectm)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(preset-lab-worker worker.cpp)
target_compile_features(preset-lab-worker PRIVATE cxx_std_17)
target_compile_definitions(preset-lab-worker PRIVATE GL_SILENCE_DEPRECATION)
# Older archived snapshots predate this API and use the private TimeKeeper hook instead.
file(READ "${PROJECTM_SOURCE}/src/libprojectM/ProjectM.hpp" PROJECTM_HEADER)
string(FIND "${PROJECTM_HEADER}" "SetFrameTime(" PROJECTM_FRAME_TIME_API)
if(NOT PROJECTM_FRAME_TIME_API EQUAL -1)
    target_compile_definitions(preset-lab-worker PRIVATE PRESET_LAB_HAS_FRAME_TIME)
endif()
target_include_directories(preset-lab-worker PRIVATE "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor/glm")
target_link_libraries(preset-lab-worker PRIVATE libprojectM::projectM SDL2::SDL2 OpenGL::GL)

if(CATALOG_TV)
 target_compile_definitions(preset-lab-worker PRIVATE CATALOG_TV)
endif()
