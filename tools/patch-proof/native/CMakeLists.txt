cmake_minimum_required(VERSION 3.22)
project(patch_proof VERSION 4.2.0 LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_GLM OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_GLES ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_FLEX ON)
set(CMAKE_DISABLE_FIND_PACKAGE_BISON ON)
add_subdirectory("${PROJECTM_SOURCE}" projectm)
add_executable(patch-proof-worker worker.cpp)
target_compile_features(patch-proof-worker PRIVATE cxx_std_17)
target_compile_definitions(patch-proof-worker PRIVATE PRESET_LAB_HAS_FRAME_TIME USE_GLES)
if(PATCH_PROOF_TV)
 target_compile_definitions(patch-proof-worker PRIVATE PATCH_PROOF_TV)
endif()
target_include_directories(patch-proof-worker PRIVATE "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor/glm" "${PROJECTM_SOURCE}/vendor/glad/include" "${PROJECTM_SOURCE}/vendor/projectm-eval/projectm-eval/api")
target_link_libraries(patch-proof-worker PRIVATE libprojectM::projectM glm::glm EGL GLESv3)
