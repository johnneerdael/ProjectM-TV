cmake_minimum_required(VERSION 3.22)
project(preset_lab VERSION 4.1.7 LANGUAGES C CXX)
if(NOT PROJECTM_SOURCE)
    message(FATAL_ERROR "PROJECTM_SOURCE must identify the isolated instrumented engine")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL_UI OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_FLEX ON)
set(CMAKE_DISABLE_FIND_PACKAGE_BISON ON)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_CURRENT_SOURCE_DIR}/analysis_hooks.hpp>")
add_subdirectory("${PROJECTM_SOURCE}" projectm)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(preset-lab-worker worker.cpp)
target_compile_features(preset-lab-worker PRIVATE cxx_std_17)
target_compile_definitions(preset-lab-worker PRIVATE GL_SILENCE_DEPRECATION)
target_include_directories(preset-lab-worker PRIVATE "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor/glm")
target_link_libraries(preset-lab-worker PRIVATE libprojectM::projectM SDL2::SDL2 OpenGL::GL)
